cmake_minimum_required(VERSION 3.22)
project(StudentMemoryCopilot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(memory_core source/sim/Engines.cpp source/sim/ImportEngine.cpp source/app/AppState.cpp source/services/FakeBackend.cpp)
target_include_directories(memory_core PUBLIC source)
target_include_directories(memory_core SYSTEM PUBLIC third_party)
set_target_properties(memory_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(memory_core PRIVATE /W4 /WX)
else()
  target_compile_options(memory_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ANDROID)
  add_library(memory_jni SHARED platform/android/jni/Bridge.cpp)
  target_link_libraries(memory_jni PRIVATE memory_core)
else()
  enable_testing()
  add_executable(core_tests tests/CoreTests.cpp)
  target_link_libraries(core_tests PRIVATE memory_core)
  add_test(NAME core_tests COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR}/data)
  add_executable(memory_demo source/app/main.cpp)
  target_link_libraries(memory_demo PRIVATE memory_core)
endif()
